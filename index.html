import json import pandas as pd from pathlib import Path excel_path = Path("HO_Top10Company by Parish.xlsx") html_template_path = Path("dashboard_template.html") output_path = Path("index.html") df = pd.read_excel(excel_path, sheet_name="ALL") numeric_columns = [ "Total Policies In-Force", "Total Policies with Wind Inclusion", "Total Policies with Wind Exclusion", "Total Exposures with Wind Inclusion", "Total Exposures with Wind Exclusion", "Total Direct Written Premium with Wind Inclusion", "Total Direct Written Premium with Wind Exclusion", ] for column in numeric_columns: df[column] = pd.to_numeric(df[column], errors="coerce").fillna(0) df["Parish"] = df["Parish"].astype(str).str.strip() df["Company Name"] = df["Company Name"].astype(str).str.strip() # Combine duplicate parish-company records. summary = ( df.groupby(["Parish", "Company Name"], as_index=False)[numeric_columns] .sum() ) summary["Average Exposure per Policy"] = ( summary["Total Exposures with Wind Inclusion"] .div(summary["Total Policies In-Force"].replace(0, pd.NA)) .fillna(0) ) # Select the top 11 companies in each parish by Total Policies In-Force. top_11 = ( summary.sort_values( ["Parish", "Total Policies In-Force"], ascending=[True, False] ) .groupby("Parish", group_keys=False) .head(11) ) parish_data = {} for parish, group in top_11.groupby("Parish"): group = group.sort_values( "Total Policies In-Force", ascending=False ) parish_data[parish] = { "company": group["Company Name"].tolist(), "policies": group["Total Policies In-Force"].astype(int).tolist(), "wind_in": group[ "Total Policies with Wind Inclusion" ].astype(int).tolist(), "wind_ex": group[ "Total Policies with Wind Exclusion" ].astype(int).tolist(), "exposure_in": group[ "Total Exposures with Wind Inclusion" ].tolist(), "exposure_ex": group[ "Total Exposures with Wind Exclusion" ].tolist(), "premium_in": group[ "Total Direct Written Premium with Wind Inclusion" ].tolist(), "premium_ex": group[ "Total Direct Written Premium with Wind Exclusion" ].tolist(), "avg_exposure": group[ "Average Exposure per Policy" ].tolist(), } template = html_template_path.read_text(encoding="utf-8") updated_html = template.replace( "const parishData = {};", "const parishData = " + json.dumps(parish_data, separators=(",", ":")) + ";" ) output_path.write_text(updated_html, encoding="utf-8") print("Updated index.html created successfully.")
